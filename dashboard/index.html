Install

npm install rhenium-contracts
forge install ahmardchain/Rhenium

Also needs reactive-lib-omni with the @reactive/ remapping (see the README).

Integrate in 10 lines

// Reactive Lasna: instead of AbstractReactive
contract MyReactive is RheniumReactive {
    function _react(LogRecord calldata log_) internal override {
        _sendTracked(1301, myCallback, 200_000, abi.encode(log_.topic1));
    }
}
// Destination chain: instead of AbstractCallback
contract MyCallback is RheniumCallback {
    function _onDelivery(uint256 receiptId, bytes calldata data)
        internal override { /* your action */ }
}
